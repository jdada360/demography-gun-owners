3.2 Define post treatment

These variables are defined so that they can be used with the did package, which uses the Callaway and Sant’anna (2021) estimator.

Code 
clean <-
  merge %>% 
  mutate(
    across(
      ends_with("date"),
      ~ year(.x),
      .names = "{str_replace_all(.col, 'date', 'year')}"
    ),
    across(
      ends_with("_year"),
      ~ case_when(
        !is.na(.x) ~ year >=.x,
        TRUE ~ F
      ) %>% as.numeric() %>% 
        factor(
          levels = 0:1,
          labels = c("Control", "Treatment")
        ),
      .names = "{str_replace_all(.col, 'year', 'post')}"
    ),
    # compute adjusted years so that implementation dates align with survey dates
    across(
      ends_with("_year"),
      ~ case_when(
        is.na(.x) ~ NA_real_,
        TRUE ~ .x + (.x %% 2 == 0)
      ),
      .names = "{str_replace_all(.col, 'year', 'year_adj')}"
    )
  ) %>% 
  group_by(year) %>% 
  mutate(t = cur_group_id()) %>% 
  ungroup() %>% 
  mutate(
     across(
      ends_with("_year_adj"),
      ~ case_when(
        .x == year ~ t,
        TRUE ~ NA_real_
      ),
      .names = "{str_replace_all(.col, 'year_adj', 'g')}"
    )
  ) %>% 
  group_by(state)  %>% 
   mutate(
    across(
      ends_with("g"), 
      ~ max(ifelse(is.na(.x), 0, max(.x, na.rm = T))) %>% as.numeric
      ),
    stateid = cur_group_id()
  )  %>% 
  group_by(state, year) %>% 
  mutate(cluster = cur_group_id()) %>% 
  ungroup()

3.3 Check if we want a balanced panel

Balanced = Keep states that we observe for all 7 years.

Code 
balanced <-
  clean %>% 
  group_by(state) %>% 
  filter(n() == 7) %>% 
  ungroup()
