This code currently does not include any control variables.

5.2 Simple OLS

Code 
simple <-
  map(
    names(select(panel, ends_with("post"))),
    ~ {
      fml <- 
        paste0("gun ~ ", .x, "| state + year")  %>% 
        as.formula()
      
      fixest::feols(
          fml,
          cluster = ~ cluster,
          data = panel
        )
    }
  )

5.3 Run DiD

Code 
y_list <-  names(select(panel, ends_with("_g")))
Code 
att <-
  map(
    y_list,
    ~ att_gt(
      yname = "gun",
      tname = "t",
      idname = "stateid",
      gname = .x,
      data = balanced,
      print_details = FALSE,
      cores = 4,
      clustervars = "stateid",
      control_group = "notyettreated"
      )
  )

5.4 Aggregate Group-Time Average Treatment Effects

No significant weighted average of all group-time average treatment effects.

Code 
map(
  att,
  ~ aggte(
    .x, 
    type = "simple",
    na.rm = TRUE
    ) %>% tidy()
  ) %>% 
  bind_rows() %>% 
  mutate(
    policy = 
      y_list %>% 
      str_replace_all(
        c(
          "any_g" = "Any Sports Betting",
          "retail_g" = "Retail",
          "online_g" = "Online",
          "igaming_g" = "iGaming"
        )
      )
    ) %>% 
  ggplot(
    aes(
      x = estimate,
      xmin = conf.low,
      xmax = conf.high,
      y = policy
    )
  ) +
  geom_point() +
  geom_errorbarh(width = 0.5) +
  geom_vline(xintercept = 0, linetype = "dashed") +
  theme_event +
  labs(
    x = "Share of Gun Owners",
    y = "Legalization"
  )
