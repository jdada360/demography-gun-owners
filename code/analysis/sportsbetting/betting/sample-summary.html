4.1 Sample sizes

Code 
counts <- 
  balanced %>% 
    group_by(state) %>% 
    summarise(
       across(ends_with("_year"), max),
       n = n(),
       .groups = "drop"
    ) %>% 
  arrange(any_year) %>% 
    mutate(
      across(
        - c(state, n),
        ~ case_when(
          is.na(.x) ~ "-",
          TRUE ~ as.character(.x)
        )
      )
    ) %>% 
    bind_rows(
      balanced %>%
        mutate(state = "Overall (average)") %>% 
        group_by(state) %>% 
        summarise(
          across(
            ends_with("_year"), 
            ~ sprintf("%.0f", (mean(.x, na.rm = T)))
          ),
          n = n()
        )
    )

4.3 State averages

Code 
balanced %>% 
  bind_rows(balanced %>% mutate(state = "Overall")) %>% 
  group_by(state) %>% 
  summarise(
    across(
      gun,
      list(
        mean = ~ mean(.x),
        conf.low = ~ mean(.x) - 1.96 * sd(.x)/sqrt(7), # 7 observations per state
        conf.high = ~ mean(.x) + 1.96 * sd(.x)/sqrt(7) 
      ),
      .names = "{.fn}"
    )
  ) %>% 
  ggplot(
    aes(x = reorder(state, mean),
        y = mean,
        ymin = conf.low,
        ymax = conf.high)
  ) +
  geom_point() +
  geom_errorbar() +
  theme_event +
  scale_y_continuous(breaks = seq(0.1, 0.4, 0.05)) +
  labs(
    x = "State",
    y = "Share of gun owners"
  ) +
  coord_flip()
